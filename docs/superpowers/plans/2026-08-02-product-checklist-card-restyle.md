### Task 1: Add the Visual-Style Regression Test

**Files:**

- [ ] **Step 1: Add a failing card-style test**

Append this test:

```js
test('PDP symptom checklist uses white bordered cards with blue checks', async ({ page }) => {
  await page.setViewportSize({ width: 1440, height: 900 });
  await page.goto('file://' + path.join(SITE_DIR, 'product.html'));
  await page.evaluate(() => document.fonts.ready);

  const styles = await page.locator('.pf-checklist-band .pf-checklist li').first().evaluate((item) => {
    const card = window.getComputedStyle(item);
    const check = window.getComputedStyle(item, '::before');
    const band = window.getComputedStyle(item.closest('.pf-checklist-band'));
    return {
      bandBackground: band.backgroundColor,
      cardBackground: card.backgroundColor,
      borderStyle: card.borderStyle,
      borderWidth: card.borderWidth,
      borderRadius: parseFloat(card.borderRadius),
      minHeight: parseFloat(card.minHeight),
      checkColor: check.color,
      checkContent: check.content,
    };
  });

  expect(styles.bandBackground).not.toBe(styles.cardBackground);
  expect(styles.cardBackground).toBe('rgb(251, 253, 251)');
  expect(styles.borderStyle).toBe('solid');
  expect(parseFloat(styles.borderWidth)).toBeGreaterThan(0);
  expect(styles.borderRadius).toBeGreaterThanOrEqual(8);
  expect(styles.minHeight).toBeGreaterThanOrEqual(64);
  expect(styles.checkColor).toBe('rgb(37, 150, 190)');
  expect(styles.checkContent).toContain('✓');
});
```
